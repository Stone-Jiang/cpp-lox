// Inherited method.
class Animal {
  speak() {
    print "animal";
  }
}

class Dog < Animal {}
Dog().speak(); // animal

// Overridden methods dispatch according to the instance's class.
class Cat < Animal {
  speak() {
    print "cat";
  }
}

Cat().speak();

fun makeSpeak(animal) {
  animal.speak();
}

makeSpeak(Dog()); // animal
makeSpeak(Cat()); // cat

// A subclass can extend an overridden method through super.
class Pastry {
  describe() {
    print "pastry";
  }
}

class Doughnut < Pastry {
  describe() {
    super.describe();
    print "doughnut";
  }
}

Doughnut().describe();
// pastry
// doughnut

// The inherited initializer sets state used by an overridden method.
class Named {
  init(name) {
    this.name = name;
  }

  describe() {
    print this.name;
  }
}

class Employee < Named {
  describe() {
    print "employee";
    super.describe();
  }
}

var employee = Employee("Ada");
employee.describe();
// employee
// Ada
